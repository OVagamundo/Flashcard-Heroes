func play_parallel_management_events(start_snapshot: Dictionary, events: Array[CombatEvent]) -> void:
	# Ensure position snapshot is set for these events
	_position_snapshot = start_snapshot
	
	var death_events: Array[CombatEvent] = []
	var has_damage = false
	var has_buff = false
	var has_heal = false
	
	# For parallel events, we fire off the animations concurrently,
	# BUT we defer DEATH events until after damage/buff animations complete.
	for event in events:
		# Process trinket bounces first
		_play_trinket_activations_for_event(event)
		
		if event.type == CombatEvent.Type.LOG_MESSAGE:
			continue
			
		if event.type == CombatEvent.Type.DEATH:
			death_events.append(event)
			continue
			
		var anim = null
		match event.type:
			CombatEvent.Type.BUFF:
				has_buff = true
				anim = AnimationRegistry.get_animation("buff")
				Audio.play_sfx("combat_buff")
			CombatEvent.Type.HEAL:
				has_heal = true
				anim = AnimationRegistry.get_animation("heal")
				Audio.play_sfx("combat_heal")
			CombatEvent.Type.DAMAGE:
				has_damage = true
				anim = AnimationRegistry.get_animation("damage")
			CombatEvent.Type.SUMMON:
				anim = AnimationRegistry.get_animation("summon")
				
		if anim:
			# Fire and forget (do not await)
			anim.execute(self, event.target_uuids, event.visual_payload)
			
	var base_delay = 0.0
	if has_damage:
		base_delay = 1.0 # Give time for projectile and impact
	elif has_buff or has_heal:
		base_delay = 0.6 # Normal buff popups
		
	if base_delay > 0:
		await get_tree().create_timer(base_delay).timeout
		
	if not death_events.is_empty():
		for event in death_events:
			var anim = AnimationRegistry.get_animation("death")
			if anim:
				anim.execute(self, event.target_uuids, event.visual_payload)
		
		# Wait for death fades to finish (death duration is usually ~0.8s)
		await get_tree().create_timer(1.0).timeout

func acquire_unit_lock(uuid: String) -> void:
	while _unit_locks.has(uuid):
		await get_tree().create_timer(0.05).timeout
	_unit_locks[uuid] = true

func release_unit_lock(uuid: String) -> void:
	_unit_locks.erase(uuid)

func _get_all_targets_for_chain(events: Array[CombatEvent]) -> Array[String]:
	var targets: Dictionary = {}
	for event in events:
		if not event.source_uuid.is_empty():
			targets[event.source_uuid] = true
		for t in event.target_uuids:
			targets[t] = true
	var arr: Array[String] = []
	for k in targets.keys():
		arr.append(k)
	return arr

func play_locked_management_chain(snapshot: Dictionary, events: Array[CombatEvent]) -> void:
	var target_uuids = _get_all_targets_for_chain(events)
	target_uuids.sort() # Sort to prevent deadlocks
	
	for target in target_uuids:
		await acquire_unit_lock(target)
		
	# Now that we have exclusive locks on all units involved in this chain, play it sequentially
	await _animate_events(events)
	
	for target in target_uuids:
		release_unit_lock(target)

func play_parallel_management_events(snapshot: Dictionary, events: Array[CombatEvent]) -> void:
	# Clear any stale visual positions so the dynamic fallback correctly searches the live scene tree
	_position_snapshot.clear()
	
	# Fire and forget the locked management chain
	play_locked_management_chain(snapshot, events)
